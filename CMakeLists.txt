cmake_minimum_required(VERSION 3.16)
project(Tunuino LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(tunuino_engine_math STATIC firmware/core/engine_math.c)
target_include_directories(tunuino_engine_math PUBLIC firmware/core)
if(NOT CMAKE_CROSSCOMPILING)
    enable_testing()
    add_executable(test_engine_math tests/test_engine_math.c)
    target_link_libraries(test_engine_math PRIVATE tunuino_engine_math m)
    add_test(NAME engine_math COMMAND test_engine_math)
endif()
