# Consolidated host coverage: one CMake/CTest job instead of per-feature workflows.
function(tunuino_add_host_test name)
    set(sources "tests/${name}.c")
    foreach(module IN LISTS ARGN)
        if(module STREQUAL "console_tx_queue" OR module STREQUAL "console_uart_pump" OR module STREQUAL "console_protocol")
            list(APPEND sources "firmware/console/${module}.c")
        elseif(module MATCHES "^platform:")
            string(REPLACE "platform:" "" module "${module}")
            list(APPEND sources "firmware/platform/stm32f429/${module}.c")
        else()
            list(APPEND sources "firmware/core/${module}.c")
        endif()
    endforeach()
    add_executable(${name} ${sources})
    target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -pedantic)
    add_test(NAME ${name} COMMAND ${name})
endfunction()
