cmake_minimum_required(VERSION 3.21)
project(TunuinoConsole VERSION 0.1.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets SerialPort Network)
qt_standard_project_setup()
add_library(TunuinoProtocol STATIC src/protocol/FrameCodec.cpp src/protocol/FrameCodec.h)
target_link_libraries(TunuinoProtocol PUBLIC Qt6::Core)
qt_add_executable(TunuinoConsole WIN32 resources.qrc src/main.cpp src/MainWindow.cpp src/MainWindow.h src/logging/GraphLogWidget.cpp src/logging/GraphLogWidget.h src/logging/LogData.h src/logging/GraphLogLayout.h src/ecu/SerialSession.cpp src/ecu/SerialSession.h src/ecu/AutoConnection.cpp src/ecu/AutoConnection.h src/ecu/EcuPowerState.h)
if(NOT DEFINED TUNUINO_BUILD_VERSION)
    set(TUNUINO_BUILD_VERSION "${PROJECT_VERSION}")
endif()
target_compile_definitions(TunuinoConsole PRIVATE TUNUINO_CONSOLE_VERSION="${TUNUINO_BUILD_VERSION}")
if(WIN32)
    target_sources(TunuinoConsole PRIVATE windows_icon.rc)
    target_link_libraries(TunuinoConsole PRIVATE setupapi winusb)
endif()
target_link_libraries(TunuinoConsole PRIVATE Qt6::Widgets Qt6::SerialPort Qt6::Network TunuinoProtocol)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(FrameCodecTest tests/FrameCodecTest.cpp)
    target_link_libraries(FrameCodecTest PRIVATE TunuinoProtocol Qt6::Test)
    add_test(NAME FrameCodecTest COMMAND FrameCodecTest)
    add_executable(LogDataTest tests/LogDataTest.cpp)
    target_include_directories(LogDataTest PRIVATE src)
    add_test(NAME LogDataTest COMMAND LogDataTest)
    qt_add_executable(AutoConnectionTest tests/AutoConnectionTest.cpp src/ecu/AutoConnection.cpp)
    target_include_directories(AutoConnectionTest PRIVATE src)
    target_link_libraries(AutoConnectionTest PRIVATE Qt6::Core)
    add_test(NAME AutoConnectionTest COMMAND AutoConnectionTest)
    qt_add_executable(GraphLogWidgetTest tests/GraphLogWidgetTest.cpp src/logging/GraphLogWidget.cpp)
    target_include_directories(GraphLogWidgetTest PRIVATE src)
    target_link_libraries(GraphLogWidgetTest PRIVATE Qt6::Widgets Qt6::Test)
    add_test(NAME GraphLogWidgetTest COMMAND GraphLogWidgetTest)
    set_tests_properties(GraphLogWidgetTest PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS TunuinoConsole RUNTIME DESTINATION .)
qt_generate_deploy_app_script(TARGET TunuinoConsole OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script})

# Read-only serial identity probe for physical ECU validation on Windows.
qt_add_executable(EcuIdentityProbe tests/EcuIdentityProbe.cpp src/ecu/SerialSession.cpp src/ecu/SerialSession.h)
target_link_libraries(EcuIdentityProbe PRIVATE Qt6::SerialPort TunuinoProtocol)
