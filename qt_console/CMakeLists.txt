cmake_minimum_required(VERSION 3.21)
project(TunuinoConsole VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets SerialPort)
qt_standard_project_setup()
add_library(TunuinoProtocol STATIC src/protocol/FrameCodec.cpp src/protocol/FrameCodec.h)
target_link_libraries(TunuinoProtocol PUBLIC Qt6::Core)
qt_add_executable(TunuinoConsole WIN32 src/main.cpp src/MainWindow.cpp src/MainWindow.h)
target_link_libraries(TunuinoConsole PRIVATE Qt6::Widgets Qt6::SerialPort TunuinoProtocol)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(FrameCodecTest tests/FrameCodecTest.cpp)
    target_link_libraries(FrameCodecTest PRIVATE TunuinoProtocol Qt6::Test)
    add_test(NAME FrameCodecTest COMMAND FrameCodecTest)
endif()
install(TARGETS TunuinoConsole RUNTIME DESTINATION .)
qt_generate_deploy_app_script(TARGET TunuinoConsole OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script})
